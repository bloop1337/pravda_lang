
# Правда - это компилируемый, строго типизированый, структурный язык программирования c ключевыми словами на русском.

### Допустимые симовлы в коде:
    "а-я,0-9", "(", ")", "{", "}", "+", "-", "=", "*", """, ";",
    ":", ">" "<", "_", ".", "," "%",

### Зарезервированные слова:
    фн, если, либо, иначе, пока, или, и, включи, конст,
    верни, сломай, пропусти, прыгни, структ.

### Примитивные типы данных:
ничего - пустота, аля void, но не с си, там он мощный(если указатель)
цел    - 32 битное знаковое целое число                ( int)
нат    - 32 битное беззнаковое целое число             (uint)
дробь  - 32 битное знаковое число с плавающей точкой   (float)
длин   - 64 битное знаковое целое число                (long)
точн   - 64 битное знаковое число с плавающей точкой   (double)
символ - utf-8 символ.
лог    - 1 битное логическое значение истина/ложь      (bool)

### Пример кода на этом языке:
```
включи "стандарт";

фн главная(): цел {
    печатай("Привет, Мир!\н");
    верни 0;
}
```

### Быстрый вход в язык

#### Коментарии
```
% это коментарий, компилятор игнорирует его.
```

#### Условия
```
если условие {
    % будет выполнен код внутри скобок
    % если условие истина
}
либо другое_условие {
    % будет выполнен код внутри скобок
    % если условие ложь, и другое_условие истина
}
иначе {
    % ни одно условие выше не выполнено,
    % будет выполнен код внутри этих скобок
}
```

```
    первое_условие: лог = истина;
    второе_условие: лог = ложь;
    
    % есть в языке логические операторы
    если первое_условие и второе_условие {
        % код в скобках(здесь) не выполнится, так как второе_условие ложь
    } 

    если первое_условие или второе_условие {
        % код выполнится, так как первое_условие истинна 
    }

    если не(второе_условие) {
        % код выполнится, так как второе_условие становится истинной
        % не - функция из стандартной библиотеки
    }
```


#### Цикл
```
    пока условие {
        % пока условие истина 
        % код в скобках будет выполнятся снова и снова
    }
```

```
    % данная программа выведет текст 10 раз подрят
    счетчик: нат = 0;
    пока счетчик < 10 {
        печатай("Мы в цикле, счетчик равен: ?\н", счетчик);
        счетчик = счетчик + 1;
    }
```

```
    % данная программа выведет текст 10 раз подрят
    счетчик: нат = 0;
    пока счетчик < 10 {
        печатай("Мы в цикле, счетчик равен: ?\н", счетчик);
        счетчик = счетчик + 1;
    }
```

```
    загаданное_число: нат = загадай_до(10);
    ввод_человека: нат = 0; 

    пока истина {
         печатай("Угадай число от 0 до 10: ");
         ввод_человека = прочитай() как нат;

         если ввод_человека == загаданное_число {
            печатай("Молодец, угадал\н");
            сломай;
         } иначе {
             печатай("Не угадал, попробуй еще раз\н");
         }
    }

```

#### Функции
```
    % объявление функции, которая ничего не возвращает
    фн поздоровайся() {
        печатай("Здраствуйте!");
    }
    % вызов
    поздоровайся();
```
```
    % объявление функции, которая складывает два передаваемых числа
    фн сложи(а: цел, б: цел): цел {
        верни а + б;
    }
    % вызов
    результат: цел = сложи(2, 10);
```
